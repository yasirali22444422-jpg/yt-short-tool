Rule 4 — BACKGROUND / PROP LOCK:
\- If the student specifies the background or ground props, preserve them exactly

\- Use the student's lighting if provided
\- Otherwise infer clean commercial ASMR lighting that suits the products and background

function setupToggle(groupId, singleSelect = true, accentClass = '') {
  const group = document.getElementById(groupId);
  if (!group) return;
  group.querySelectorAll('.tog').forEach(btn => {
    btn.addEventListener('click', function() {
      if (singleSelect) {
        group.querySelectorAll('.tog').forEach(x => x.classList.remove('on', 'teal', 'amber', 'blue'));
      }
      this.classList.add('on');
      if (accentClass) this.classList.add(accentClass);
    });
  });
}

['sourceModeGroup','genreGroup','langGroup','voYNGroup','handGroup','cameraGroup','introGroup','holdCamGroup','prodModeGroup','mystGroup','toyGroup','strictGroup'].forEach(id => setupToggle(id));

function getActive(groupId) {
  const el = document.querySelector('#' + groupId + ' .tog.on');
  return el ? el.dataset.val : '';
}

function updateVoiceoverState() {
  document.getElementById('voDetails').style.maxHeight = getActive('voYNGroup') === 'yes' ? '340px' : '0';
}

function updateIntroState() {
  const enabled = getActive('introGroup') === 'yes';
  document.getElementById('introDetails').style.maxHeight = enabled ? '220px' : '0';
}

function updateProductMode() {
  const isAI = getActive('prodModeGroup') === 'ai';
  document.getElementById('aiProdWrap').style.display = isAI ? 'block' : 'none';
  document.getElementById('manualProdWrap').style.display = isAI ? 'none' : 'block';
}

['voYNGroup','introGroup','prodModeGroup'].forEach(id => {
  document.getElementById(id).addEventListener('click', () => {
    updateVoiceoverState();
    updateIntroState();
    updateProductMode();
  });
});
